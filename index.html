""" mPOS (Mobile Point of Service) - Core Settlement Engine Prototype Version: 1.1.0 Author: Vershima Kwaghshi DESCRIPTION: This engine simulates the cross-border settlement infrastructure for mPOS. It integrates the following architectural pillars: 1. Digital Passport (DP) Identity Verification. 2. Real-Time FX Rate Locking. 3. The 0.2% Collective Stability Buffer. 4. The 72-Hour Velocity Rule (Escalating Fees). 5. Randomized Agent Routing to prevent collusion. """ import uuid import random import json from datetime import datetime, timedelta class MPOSEngine: def __init__(self): # Operational Constants self.CORRIDOR_STABILITY_BUFFER = 0.002 # 0.2% self.VELOCITY_FEE_STANDARD = 0.0005 # 0.05% (< 3 Days) self.VELOCITY_FEE_PENALTY = 0.0008 # 0.08% (> 3 Days) self.ONBOARDING_FEE = 1.29 # USD # System State self.passport_registry = {} self.buffer_reserve_pool = 0.0 self.total_revenue = 0.0 self.ledger = [] # --- IDENTITY LAYER --- def issue_digital_passport(self, entity_name, region): """Creates a Digital Passport (DP) required for any engine activity.""" dp_id = f"DP-{uuid.uuid4().hex[:12].upper()}" self.passport_registry[dp_id] = { "entity": entity_name, "region": region, "reputation": 100, "active": True } self.total_revenue += self.ONBOARDING_FEE return dp_id def _verify_passport(self, dp_id): return dp_id in self.passport_registry and self.passport_registry[dp_id]["active"] # --- SETTLEMENT LAYER --- def _get_routing_agent(self): """Randomizes the payout agent to enforce neutrality.""" agents = ["AG-NORTH-TX", "AG-SOUTH-TX", "AG-EAST-TX", "AG-WEST-TX"] return random.choice(agents) def process_transaction(self, dp_id, amount_source, rate, delay_days=1): """ The Core mPOS Settlement Logic. Converts source principal to destination ledger balance with buffer retention. """ if not self._verify_passport(dp_id): return {"status": "REJECTED", "reason": "INVALID_IDENTITY"} # Step 1: Gross Value & Buffer Retention gross_dest = amount_source * rate buffer_amount = gross_dest * self.CORRIDOR_STABILITY_BUFFER self.buffer_reserve_pool += buffer_amount # Step 2: Velocity Fee Assessment (The 72-Hour Rule) net_balance = gross_dest - buffer_amount if delay_days <= 3: fee_rate = self.VELOCITY_FEE_STANDARD logic = "STANDARD_SETTLEMENT" else: fee_rate = self.VELOCITY_FEE_PENALTY logic = "STAGNATION_AUTO_CONVERSION" fee_amount = net_balance * fee_rate self.total_revenue += fee_amount # Step 3: Final Payout and Ledger Commit final_payout = net_balance - fee_amount tx_id = f"MPOS-{uuid.uuid4().hex[:14].upper()}" record = { "tx_id": tx_id, "timestamp": datetime.now().isoformat(), "origin_dp": dp_id, "payout_agent": self._get_routing_agent(), "metrics": { "source_amount": amount_source, "fx_rate": rate, "gross_dest": round(gross_dest, 2), "buffer_retained": round(buffer_amount, 2), "service_fee": round(fee_amount, 2), "net_payout": round(final_payout, 2) }, "logic": logic } self.ledger.append(record) return record # --- MONITORING LAYER --- def get_infrastructure_stats(self): """Returns the current state of the mPOS corridor.""" return { "active_passports": len(self.passport_registry), "collective_buffer_reserve": round(self.buffer_reserve_pool, 2), "engine_total_revenue": round(self.total_revenue, 2), "transaction_count": len(self.ledger) } # --- EXAMPLE USAGE --- if __name__ == "__main__": # Initialize Infrastructure mpos = MPOSEngine() # 1. Onboard Participating Bank bank_dp = mpos.issue_digital_passport("Global Corridor Bank", "NG-UK") # 2. Simulate High-Volume Settlement (Processed Day 1) # Principal: 10,000 GBP, Rate: 1,850 tx1 = mpos.process_transaction(bank_dp, 10000, 1850, delay_days=1) # 3. Simulate Delayed Settlement (Processed Day 5 - Penalty) tx2 = mpos.process_transaction(bank_dp, 5000, 1850, delay_days=5) # 4. Display Results print("\n--- TRANSACTION LOG (SAMPLE) ---") print(json.dumps(tx1, indent=4)) print("\n--- INFRASTRUCTURE SUMMARY ---") print(json.dumps(mpos.get_infrastructure_stats(), indent=4))
